import React, { useCallback, useMemo, useState, useEffect } from 'react';
import { useRoomState } from './useRoomState';
import { RoomList } from './RoomList';
import { RoomDetails } from './RoomDetails';
import { WebSocketMessage } from './types';

// -----------------------------
// Step 1: Custom Hook for State Logic
// -----------------------------
export function useRoomState(initialState: any) {
  const [roomData, setRoomData] = useState(initialState);
  const [websocketMessages, setWebSocketMessages] = useState<WebSocketMessage[]>([]);

  // WebSocket connection setup
  useEffect(() => {
    const ws = new WebSocket('wss://example.com/rooms');

    ws.onmessage = (event) => {
      const message: WebSocketMessage = JSON.parse(event.data);
      setWebSocketMessages((prev) => [...prev, message]);
    };

    return () => {
      ws.close();
    };
  }, []);

  // Derived state: Filtered room data based on WebSocket messages
  const filteredRoomData = useMemo(() => {
    return roomData.filter((room) => {
      // Example: Filter rooms based on WebSocket messages
      return websocketMessages.some((msg) => msg.roomId === room.id);
    });
  }, [roomData, websocketMessages]);

  return {
    roomData,
    setRoomData,
    filteredRoomData,
    websocketMessages,
  };
}

// -----------------------------
// Step 2: Memoized Child Components
// -----------------------------
const RoomList = React.memo(({ rooms }: { rooms: any[] }) => {

    </div>
  );
});

    <div>

});
