```typescript
/**
 * Calculates the appropriate synchronization action based on the time drift
 * between client and server.
 * 
 * @param clientTime - The current time on the client side
 * @param serverTime - The current time on the server side
 * @param drift - The absolute time difference between client and server
 * @returns A string indicating the action: 'ignore', 'adjust_rate', or 'seek'
 */
function calculateSyncAction(clientTime: number, serverTime: number, drift: number): string {
    if (drift < 250) {
        return 'ignore';
    }
    if (drift < 800) {
        return 'adjust_rate';
    }
    return 'seek';
}
```
