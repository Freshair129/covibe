function deepMerge<T>(obj1: T, obj2: T): T {
  if (typeof obj1 !== 'object' || typeof obj2 !== 'object') {
    return obj1;
  }

  const result = { ...obj1 };

  for (const key in obj2) {
    if (Object.prototype.hasOwnProperty.call(obj2, key)) {
      if (key === '$deepMerge') continue;

      if (typeof obj2[key] === 'object' && typeof result[key] === 'object') {
        const merged = deepMerge(result[key], obj2[key]);
        for (const k in merged) {
          if (Object.prototype.hasOwnProperty.call(merged, k)) {
            result[k] = merged[k];
          }
        }
      } else {
        result[key] = obj2[key];
      }
    }
  }

  return result;
}

// Example usage:
const obj1 = { a: 1, b: { x: 10, y: 20 } };
const obj2 = { b: { y: 30 }, c: 3 };

console.log(deepMerge(obj1, obj2)); // Output: { a: 1, b: { x: 10, y: 30, z: 40 }, c: 3 }
