class PriorityQueue {
  private heap: { priority: number; item: any }[];

  constructor() {
    this.heap = [];
  }

  enqueue(item: any, priority: number): void {
    // Add the new element to the end of the heap array
    const newNode = { priority, item };
    this.heap.push(newNode);
    // Bubble up to maintain heap property
    this.bubbleUp(this.heap.length - 1);
  }

  dequeue(): any {
    if (this.isEmpty()) {
      throw new Error('Queue is empty');
    }
    // Remove the root (index 0)
    const root = this.heap[0];
    // Replace root with last element
    const lastNode = this.heap.pop()!;
    if (this.heap.length > 0) {
      this.heap[0] = lastNode;
      // Bubble down to maintain heap property
      this.bubbleDown(0);
    }
    return root.item;
  }

  peek(): any {
    if (this.isEmpty()) {
      throw new Error('Queue is empty');
    }
    return this.heap[0].item;
  }

  isEmpty(): boolean {
    return this.heap.length === 0;
  }

  private bubbleUp(index: number): void {
    const parentIndex = Math.floor((index - 1) / 2);
    while (index > 0 && this.heap[index].priority < this.heap[parentIndex].priority) {
      // Swap
      [this.heap[index], this.heap[parentIndex]] = [this.heap[parentIndex], this.heap[index]];
      index = parentIndex;
      parentIndex = Math.floor((index - 1) / 2);
    }
  }

  private bubbleDown(index: number): void {
    const leftChildIndex = 2 * index + 1;
    const rightChildIndex = 2 * index + 2;
    let smallestIndex = index;

    if (leftChildIndex < this.heap.length && this.heap[leftChildIndex].priority < this.heap[smallestIndex].priority) {
      smallestIndex = leftChildIndex;

      [this.heap[index], this.heap[smallestIndex]] = [this.heap[smallestIndex], this.heap[index]];
      this.bubbleDown(smallestIndex);
    }
  }
