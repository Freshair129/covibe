function calculate_time(start_time, end_time) {
  // Assuming start_time and end_time are in a comparable format (e.g., Unix timestamps or Date objects)
  // For simplicity, let's assume they are Unix timestamps (seconds since epoch)
  if (typeof start_time === 'number' && typeof end_time === 'number') {
    return end_time - start_time;
  } else if (start_time instanceof Date && end_time instanceof Date) {
    return end_time.getTime() - start_time.getTime();
  } else {
    // Handle other potential input types if necessary, or throw an error
    throw new Error("Invalid input types. Please provide valid time representations.");
  }
}

// Example Usage:
const startTime = new Date();
// Simulate a time difference of 10 seconds
const endTime = new Date(startTime.getTime() + 10000);

try {
  const duration = calculate_time(startTime, endTime);
  console.log(`The time difference is: ${duration} milliseconds`);
} catch (error) {
  console.error(error.message);
}
