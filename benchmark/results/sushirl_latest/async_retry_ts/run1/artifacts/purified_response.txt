## Final Implementation

```typescript
interface TripRoom {
    id: string;
    hostId: string;
    passengers: string[];
    currentVideoId: string;
    isPlaying: boolean;
    serverTime: number;
}

function calculateSyncAction(clientTime: number, serverTime: number, drift: number) {
    if (drift < 250) return 'ignore';
    if (drift < 800) return 'adjust_rate';
    return 'seek';
}
```

## Usage Example

```typescript
const clientTime = 1000;
const serverTime = 1250;
const drift = Math.abs(clientTime - serverTime); // 250

const action = calculateSyncAction(clientTime, serverTime, drift);
// Returns: 'ignore' (drift is exactly 250, which is not < 250)
```
